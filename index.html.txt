<script>
function press(val) {
  document.getElementById("input").value += val;
}

function clearInput() {
  document.getElementById("input").value = "";
  document.getElementById("result").innerText = "Result will appear here";
}

function solveInput() {
  let rawInput = document.getElementById("input").value.trim();
  let input = rawInput.toLowerCase();
  let explain = document.getElementById("explainMode").checked;

  try {
    // Quadratic
    let quadMatch = input.replace(/\s+/g, '').match(/x\^2([+-]\d*)x([+-]\d+)/);

    if (quadMatch) {
      let b = quadMatch[1] === "+" || quadMatch[1] === "" ? 1 :
              quadMatch[1] === "-" ? -1 : parseFloat(quadMatch[1]);

      let c = parseFloat(quadMatch[2]);

      let d = b * b - 4 * c;

      if (d < 0) {
        document.getElementById("result").innerText = "No real solutions";
        return;
      }

      let x1 = (-b + Math.sqrt(d)) / 2;
      let x2 = (-b - Math.sqrt(d)) / 2;

      document.getElementById("result").innerText =
        explain
          ? `Quadratic:\nx = ${x1.toFixed(2)}, ${x2.toFixed(2)}`
          : `x = ${x1.toFixed(2)}, ${x2.toFixed(2)}`;

      return;
    }

    // Statistics
    if (input.includes(",")) {
      let numbers = input.split(",").map(Number).filter(n => !isNaN(n));

      if (numbers.length === 0) throw "Invalid";

      let mean = numbers.reduce((a, b) => a + b, 0) / numbers.length;

      let sorted = [...numbers].sort((a, b) => a - b);
      let mid = Math.floor(sorted.length / 2);

      let median = sorted.length % 2 !== 0
        ? sorted[mid]
        : (sorted[mid - 1] + sorted[mid]) / 2;

      let freq = {};
      numbers.forEach(n => freq[n] = (freq[n] || 0) + 1);

      let maxFreq = Math.max(...Object.values(freq));
      let mode = Object.keys(freq).filter(n => freq[n] == maxFreq);

      let variance = numbers.reduce((sum, n) => sum + Math.pow(n - mean, 2), 0) / numbers.length;
      let sd = Math.sqrt(variance);

      document.getElementById("result").innerText =
        explain
          ? `Mean = ${mean.toFixed(2)}
Median = ${median}
Mode = ${mode.join(", ")}
SD = ${sd.toFixed(2)}`
          : `Mean = ${mean.toFixed(2)}`;

      return;
    }

    // Normal calc
    let result = eval(input);
    document.getElementById("result").innerText = "Result = " + result;

  } catch {
    document.getElementById("result").innerText = "Invalid input";
  }
}
